use std::alloc::{GlobalAlloc, Layout, System};
use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
use std::time::Instant;
use rue_air::{Air, AirInst, AirInstData, AirPlace, AirPlaceBase, AirProjection, Type};
use rue_span::Span;
use serde_json::{Map, json};
use lasso::ThreadedRodeo;

struct Counting;
static ON: AtomicBool = AtomicBool::new(false);
static N: AtomicU64 = AtomicU64::new(0);
static B: AtomicU64 = AtomicU64::new(0);
#[global_allocator] static A: Counting = Counting;
unsafe impl GlobalAlloc for Counting {
    unsafe fn alloc(&self,l:Layout)->*mut u8 { let p=unsafe{System.alloc(l)}; if !p.is_null()&&ON.load(Ordering::Relaxed){N.fetch_add(1,Ordering::Relaxed);B.fetch_add(l.size() as u64,Ordering::Relaxed);} p }
    unsafe fn dealloc(&self,p:*mut u8,l:Layout){unsafe{System.dealloc(p,l)}}
    unsafe fn realloc(&self,p:*mut u8,l:Layout,s:usize)->*mut u8 {let q=unsafe{System.realloc(p,l,s)};if !q.is_null()&&ON.load(Ordering::Relaxed){N.fetch_add(1,Ordering::Relaxed);B.fetch_add(s as u64,Ordering::Relaxed);}q}
}

const FAMILIES:[(&str,usize);11]=[("match_arms",4),("call_args",2),("type_args",1),("const_values",5),("intrinsic_args",1),("block_statements",1),("struct_fields",1),("source_order",1),("array_elements",1),("enum_payload",1),("projections_and_places",3)];

fn base()->Air { let mut a=Air::new(Type::UNIT);for i in 0..64 {a.add_inst(AirInst{data:AirInstData::Const(i),ty:Type::I32,span:Span::new(0,0)});}a }
fn consume(words:&[u32],width:usize)->usize { words.chunks_exact(width).map(|v|{std::hint::black_box(v);1}).sum() }

fn main(){
 let symbols=ThreadedRodeo::new();let pool=rue_air::TypeInternPool::new();let (sid,_)=pool.register_struct(symbols.get_or_intern("P"),rue_air::StructDef{name:"P".into(),fields:Vec::new(),is_copy:false,is_linear:false,destructor:None,is_builtin:false,is_pub:false,file_id:rue_span::FileId::DEFAULT});
 let mut out=Map::new();
 for (name,width) in FAMILIES {
  let projection=name=="projections_and_places";
  let mut airs:Vec<_>=(0..128).map(|_|base()).collect();
  N.store(0,Ordering::SeqCst);B.store(0,Ordering::SeqCst);ON.store(true,Ordering::SeqCst);let start=Instant::now();
  for air in &mut airs { if projection { let projections:Vec<_>=(0..64).map(|field_index|AirProjection::Field{struct_id:sid,field_index}).collect();air.make_place(AirPlaceBase::Local(0),Type::new_struct(sid),projections); } else { let words:Vec<_>=(0..64*width).map(|i|i as u32).collect();air.add_extra(&words); } }
  let build_ns=start.elapsed().as_nanos();ON.store(false,Ordering::SeqCst);let alloc=N.load(Ordering::SeqCst);let bytes=B.load(Ordering::SeqCst);
  N.store(0,Ordering::SeqCst);ON.store(true,Ordering::SeqCst);let start=Instant::now();let mut n=0;
  for _ in 0..20000 { n+=if projection {airs[0].get_place_projections(airs[0].get_place(rue_air::AirPlaceRef::from_raw(0))).iter().map(|v|{std::hint::black_box(v);1}).sum()}else{consume(airs[0].get_extra(0,(64*width) as u32),width)}; }
  let traversal_ns=start.elapsed().as_nanos();ON.store(false,Ordering::SeqCst);
  let logical=if projection {64*std::mem::size_of::<AirProjection>()+std::mem::size_of::<AirPlace>()}else{64*width*4};
  out.insert(name.into(),json!({"elements":64,"builds":128,"build_ns":build_ns,"build_allocations":alloc,"build_allocated_bytes":bytes,"traversals":20000,"traversal_ns":traversal_ns,"traversal_allocations":N.load(Ordering::SeqCst),"elements_per_second":n as f64/(traversal_ns as f64/1e9),"side_table_logical_bytes":logical,"side_table_capacity_bytes":logical,"peak_staging_bytes":if matches!(name,"match_arms"|"const_values"|"projections_and_places"){logical}else{0}}));
 }
 println!("{}",json!({"schema_version":1,"families":out}));
}
